## create

Create a new inference provider configuration.

```python
client.inference_providers.create(
    name="OpenAI GPT-4",
    provider_type="openai",
    config={
        "api_key": "sk-...",
        "model": "gpt-4",
        "temperature": 0.7
    },
    description="Production OpenAI provider",
    is_active=True,
    project="proj_xyz789"
)
```

<ParamField path="name" type="str" required>
  Name for this inference provider
</ParamField>

<ParamField path="provider_type" type="str" required>
  Type of provider (e.g., "openai", "anthropic", "azure", "custom")
</ParamField>

<ParamField path="config" type="dict" required>
  Provider-specific configuration object (API keys, endpoints, model parameters)
</ParamField>

<ParamField path="description" type="str" optional>
  Optional description of this provider configuration
</ParamField>

<ParamField path="is_active" type="bool" optional>
  Whether this provider should be active (default: true)
</ParamField>

<ParamField path="project" type="str" optional>
  UID of the project to associate this provider with
</ParamField>

<ResponseField name="uid" type="str">
  Unique identifier for the inference provider
</ResponseField>

<ResponseField name="name" type="str">
  Name of the inference provider
</ResponseField>

<ResponseField name="provider_type" type="str">
  Type of provider
</ResponseField>

<ResponseField name="config" type="dict">
  Provider configuration
</ResponseField>
