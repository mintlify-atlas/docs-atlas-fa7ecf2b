## list

List all storage configurations.

```python
client.storage_configs.list(
    limit=10,
    cursor="abc123"
)
```

<ParamField path="limit" type="int" optional>
  Maximum number of items to return per page
</ParamField>

<ParamField path="cursor" type="str" optional>
  Pagination cursor for fetching the next page
</ParamField>

<ResponseField name="data" type="List[StorageConfig]">
  List of storage configuration objects
</ResponseField>

<ResponseField name="next_cursor" type="str | None">
  Cursor for the next page, or None if no more results
</ResponseField>

## create

Create a new storage configuration.

```python
# S3 configuration
client.storage_configs.create(
    name="Production S3",
    provider="s3",
    s3_bucket_name="my-bucket",
    s3_bucket_region="us-east-1",
    s3_access_key_id="AKIAIOSFODNN7EXAMPLE",
    s3_secret_access_key="wJalrXUtnFEMI/K7MDENG/bPxRfiCYEXAMPLEKEY"
)

# Google Cloud Storage configuration
client.storage_configs.create(
    name="GCS Bucket",
    provider="gcs",
    gc_storage_bucket_name="my-gcs-bucket",
    gc_storage_auth_json_content='{"type": "service_account", ...}'
)
```

<ParamField path="name" type="str" required>
  Name for this storage configuration
</ParamField>

<ParamField path="provider" type="str" required>
  Storage provider type (e.g., "s3", "gcs")
</ParamField>

<ParamField path="s3_bucket_name" type="str" optional>
  S3 bucket name (required for S3 provider)
</ParamField>

<ParamField path="s3_bucket_region" type="str" optional>
  S3 bucket region (e.g., "us-east-1")
</ParamField>

<ParamField path="s3_bucket_prefix" type="str" optional>
  Prefix/folder path within the S3 bucket
</ParamField>

<ParamField path="s3_access_key_id" type="str" optional>
  AWS access key ID for authentication
</ParamField>

<ParamField path="s3_secret_access_key" type="str" optional>
  AWS secret access key for authentication
</ParamField>

<ParamField path="s3_is_accelerated" type="bool" optional>
  Enable S3 transfer acceleration
</ParamField>

<ParamField path="gc_storage_bucket_name" type="str" optional>
  Google Cloud Storage bucket name (required for GCS provider)
</ParamField>

<ParamField path="gc_storage_prefix" type="str" optional>
  Prefix/folder path within the GCS bucket
</ParamField>

<ParamField path="gc_storage_auth_json_content" type="str" optional>
  JSON content of the GCS service account key
</ParamField>

<ResponseField name="uid" type="str">
  Unique identifier for the storage configuration
</ResponseField>

<ResponseField name="name" type="str">
  Name of the storage configuration
</ResponseField>

<ResponseField name="provider" type="str">
  Storage provider type
</ResponseField>
